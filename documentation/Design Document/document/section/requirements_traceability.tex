\chapter{Functional Requirements}
\section{Smartphone Application Functional Requirements}
\subsection{Account and Profile Management}

\begin{enumerate}[label=\textbf{FR\arabic*}, ref=FR\arabic*]

\item \label{fr-smartphone:register-email}
The system shall allow users to register by providing email and password


\item \label{fr-smartphone:register-external}
The system shall allow users to register using an external authentication provider

\item \label{fr-smartphone:login-email}
The system shall allow users to log in using email and password

\item \label{fr-smartphone:login-external}
The system shall allow users to log in using an external authentication provider

\item \label{fr-smartphone:reset-password}
The system shall allow users to reset their password

\item \label{fr-smartphone:create-profile}
When registration is successful, the system shall create a user profile

\item \label{fr-smartphone:default-username}
When registration is successful, the system shall assign a default username based on the user’s email

\item \label{fr-smartphone:update-username}
The system shall allow users to update their username

\item \label{fr-smartphone:update-image}
The system shall allow users to update their profile image

\item \label{fr-smartphone:update-name}
The system shall allow users to update their name

\item \label{fr-smartphone:update-surname}
The system shall allow users to update their surname

\item \label{fr-smartphone:update-birthdate}
The system shall allow users to update their birth date

\item \label{fr-smartphone:update-email}
When users are registered with email and password, the system shall allow them to update their email address

\item \label{fr-smartphone:restore-image}
When users are authenticated through an external provider, the system shall allow them to restore their profile image from the provider

\item \label{fr-smartphone:change-language}
The system shall allow users to change the application language

\item \label{fr-smartphone:pair-watch}
The system shall allow users to pair a smartwatch with their account

\item \label{fr-smartphone:weather-notifications}
The system shall allow users to enable weather notifications for selected routes

\end{enumerate}



\subsection{Home, Routes and Route Details}

\begin{enumerate}[label=\textbf{FR\arabic*}, ref=FR\arabic*]

\setcounter{enumi}{17}

\item \label{fr-smartphone:display-routes-map}
The system shall display routes on a map

\item \label{fr-smartphone:classify-routes-difficulty}
The system shall classify routes according to their difficulty level

\item \label{fr-smartphone:display-routes-color}
The system shall display routes using different colors based on their difficulty level

\item \label{fr-smartphone:route-details}
When a user selects a route, the system shall display detailed information about the selected route

\item \label{fr-smartphone:route-info}
The system shall display route information including starting point, ending point, difficulty level, length, estimated time, and elevation gain

\item \label{fr-smartphone:family-suitability}
The system shall indicate the suitability of a route for families

\item \label{fr-smartphone:route-description}
The system shall display a textual description of the route

\item \label{fr-smartphone:refreshment-points}
The system shall display refreshment points associated with the route

\item \label{fr-smartphone:route-challenges}
The system shall display challenges associated with the route

\item \label{fr-smartphone:weather-summary}
The system shall display summary weather information for the selected route

\item \label{fr-smartphone:save-route}
The system shall allow users to save a route to their profile

\item \label{fr-smartphone:remove-saved-route}
The system shall allow users to remove a saved route from their profile

\item \label{fr-smartphone:start-navigation}
The system shall allow users to start a navigation session for a selected route

\item \label{fr-smartphone:suggest-diary}
When a navigation session ends, the system shall suggest creating a diary entry for the route

\end{enumerate}




\subsection{Diary Management}

\begin{enumerate}[label=\textbf{FR\arabic*}, ref=FR\arabic*]

\setcounter{enumi}{31}


\item \label{fr-smartphone:create-diary}
The system shall allow users to create a diary entry associated with a selected route

\item \label{fr-smartphone:diary-date}
The system shall allow users to specify the date of the activity in a diary entry

\item \label{fr-smartphone:diary-duration}
The system shall allow users to specify the duration of the activity in a diary entry

\item \label{fr-smartphone:select-participants}
The system shall allow users to select followed users who participated in the activity

\item \label{fr-smartphone:diary-notes}
The system shall allow users to add textual notes to a diary entry

\item \label{fr-smartphone:refreshment-stops}
The system shall allow users to indicate the presence of refreshment stops during the activity

\item \label{fr-smartphone:completed-challenges}
The system shall allow users to select completed challenges in a diary entry

\item \label{fr-smartphone:diary-mood}
The system shall allow users to select a mood associated with the activity

\item \label{fr-smartphone:upload-images}
The system shall allow users to upload images to a diary entry

\item \label{fr-smartphone:diary-visibility}
The system shall allow users to mark a diary entry as public or private

\item \label{fr-smartphone:save-diary}
The system shall allow users to save a diary entry

\item \label{fr-smartphone:discard-diary}
The system shall allow users to discard a diary entry before saving it

\end{enumerate}

\subsection{Search and Challenges}

\begin{enumerate}[label=\textbf{FR\arabic*}, ref=FR\arabic*]

\setcounter{enumi}{43}

\item \label{fr-smartphone:search-users}
The system shall allow users to search for other users

\item \label{fr-smartphone:search-routes}
The system shall allow users to search for routes

\item \label{fr-smartphone:search-results}
The system shall display search results according to the selected search type

\item \label{fr-smartphone:user-profile-display}
The system shall display public profile information of searched users

\item \label{fr-smartphone:searched-route-info}
The system shall display route information for searched routes

\item \label{fr-smartphone:followed-diaries}
The system shall display diary entries from followed users

\item \label{fr-smartphone:available-challenges}
The system shall display the available challenges

\item \label{fr-smartphone:challenge-requirements}
The system shall display the requirements for completing each challenge

\end{enumerate}





\subsection{Profile and Friends Features}

\begin{enumerate}[label=\textbf{FR\arabic*}, ref=FR\arabic*]

\setcounter{enumi}{51}

\item \label{fr-smartphone:profile-info}
The system shall display the user’s profile picture, username, and level

\item \label{fr-smartphone:diary-count}
The system shall display the number of diary entries created by the user

\item \label{fr-smartphone:follower-count}
The system shall display the number of followers of the user

\item \label{fr-smartphone:following-count}
The system shall display the number of followed users

\item \label{fr-smartphone:access-follower-profile}
The system shall allow users to access the public profile of a follower

\item \label{fr-smartphone:access-following-profile}
The system shall allow users to access the public profile of a followed user

\item \label{fr-smartphone:share-profile}
The system shall allow users to share a public link to their profile

\end{enumerate}







\subsection{Navigation and Sensor Information}

\begin{enumerate}[label=\textbf{FR\arabic*}, ref=FR\arabic*]

\setcounter{enumi}{58}

The system shall display compass information

\item \label{fr-smartphone:heading-degrees}
The system shall display the current heading in degrees

\item \label{fr-smartphone:altitude}
The system shall display the user’s current altitude

\item \label{fr-smartphone:coordinates}

\end{enumerate}





\subsection{Weather Information}

\begin{enumerate}[label=\textbf{FR\arabic*}, ref=FR\arabic*, resume]

\item \label{fr-smartphone:fr63}
The system shall allow users to access detailed weather information for a selected route

\item \label{fr-smartphone:fr64}
The system shall display a five-day weather forecast related to the selected route

\item \label{fr-smartphone:fr65}
The system shall allow users to request weather information using their current GPS position

\item \label{fr-smartphone:fr66}
When the user selects route-based weather information, the system shall display weather data for the route area

\item \label{fr-smartphone:fr67}
When the user selects GPS-based weather information, the system shall display weather data for the user’s current position

\item \label{fr-smartphone:fr68}
The system shall display a map preview associated with the currently selected weather location

\item \label{fr-smartphone:fr69}
The system shall display weather alerts related to the currently selected weather location

\item \label{fr-smartphone:fr70}
The system shall allow users to enable weather alert notifications for a selected route

\item \label{fr-smartphone:fr71}
The system shall allow users to access a satellite map view displaying both the selected route position and the user’s current position, using distinct visual indicators

\item \label{fr-smartphone:fr72}
The system shall allow users to display weather layers on the map

\item \label{fr-smartphone:fr73}
The system shall allow users to display precipitation information on the map

\item \label{fr-smartphone:fr74}
The system shall allow users to display snow information on the map

\item \label{fr-smartphone:fr75}
The system shall allow users to display wind information on the map

\item \label{fr-smartphone:fr76}
The system shall allow users to display cloud information on the map

\item \label{fr-smartphone:fr77}
The system shall allow users to display temperature information on the map

\item \label{fr-smartphone:fr78}
The system shall allow users to display pressure information on the map

\item \label{fr-smartphone:fr79}
The system shall display a legend associated with the selected weather layer

\end{enumerate}

\subsection{Offline mode}

\begin{enumerate}[label=\textbf{FR\arabic*}, ref=FR\arabic*, resume]

\item \label{fr-smartphone:offline-detect}
The system shall detect when the internet connection is not available


\item \label{fr-smartphone:offline-page}
When the internet connection is not available, the system shall display a dedicated offline page

\item \label{fr-smartphone:offline-compass}
In offline mode, the system shall allow users to access compass functionalities

\item \label{fr-smartphone:offline-position}
In offline mode, the system shall allow users to access their current GPS position

\item \label{fr-smartphone:offline-altitude}
In offline mode, the system shall allow users to access altitude information

\item \label{fr-smartphone:offline-restore}
When the internet connection is restored, the system shall allow users to resume online functionalities

\end{enumerate}


\section{Smartwatch Application Functional Requirements}

\subsection{Access \& Synchronization}

\begin{enumerate}[label=\textbf{FR\arabic*}, ref=FR\arabic*]

\item \label{fr-watch:fr1}
The system shall allow users to access smartwatch functionalities after pairing the device with their account.

\item \label{fr-watch:fr2}
The system shall synchronize route and user data with the associated smartphone application.

\item \label{fr-watch:fr3}
When synchronization fails, the system shall notify the user.

\item \label{fr-watch:fr4}
The system shall allow users to enable weather notifications for selected routes.

\end{enumerate}


\subsection{Routes \& Map Visualization}

\begin{enumerate}[label=\textbf{FR\arabic*}, ref=FR\arabic*, resume]

\item \label{fr-watch:fr5}
The system shall display routes on a smartwatch map.

\item \label{fr-watch:fr6}
The system shall display refuge points as spots on the route.

\item \label{fr-watch:fr7}
The system shall allow users to select a refuge point on the map.

\item \label{fr-watch:fr8}
When a refuge point is selected, the system shall display associated information.

\item \label{fr-watch:fr9}
The system shall allow users to scroll through route-related information.

\item \label{fr-watch:fr10}
The system shall classify routes according to their difficulty level.

\item \label{fr-watch:fr11}
The system shall display routes using different colors based on their difficulty level.

\end{enumerate}


\subsection{Route Details}

\begin{enumerate}[label=\textbf{FR\arabic*}, ref=FR\arabic*, resume]

\item \label{fr-watch:fr12}
The system shall display key information about a selected route.

\item \label{fr-watch:fr13}
When a user selects a route, the system shall display detailed information about the selected route.

\item \label{fr-watch:fr14}
The system shall display route information including difficulty level, distance, estimated time, and elevation gain.

\item \label{fr-watch:fr15}
The system shall indicate the suitability of a route for families.

\item \label{fr-watch:fr16}
The system shall display a concise textual description of the route.

\item \label{fr-watch:fr17}
The system shall display refreshment points associated with the route.

\item \label{fr-watch:fr18}
The system shall display challenges associated with the route.

\item \label{fr-watch:fr19}
The system shall display summary weather information for the selected route.

\item \label{fr-watch:fr20}
The system shall allow users to start a navigation session for a selected route.

\end{enumerate}


\subsection{Weather Information}

\begin{enumerate}[label=\textbf{FR\arabic*}, ref=FR\arabic*, resume]

\item \label{fr-watch:fr21}
The system shall display weather information for a selected route.

\item \label{fr-watch:fr22}
The system shall display current weather conditions.

\item \label{fr-watch:fr23}
The system shall display a short-term forecast.

\item \label{fr-watch:fr24}
When weather information is unavailable, the system shall notify the user.

\item \label{fr-watch:fr25}
The system shall allow users to receive weather alert notifications for selected routes.

\item \label{fr-watch:fr26}
The system shall display weather alerts related to the currently selected weather location.

\item \label{fr-watch:fr27}
The system shall allow users to enable weather alert notifications for a selected route.

\end{enumerate}


\subsection{Friends’ Routes}

\begin{enumerate}[label=\textbf{FR\arabic*}, ref=FR\arabic*, resume]

\item \label{fr-watch:fr28}
The system shall display routes published by followed users.

\item \label{fr-watch:fr29}
The system shall allow users to browse routes shared by friends.

\item \label{fr-watch:fr30}
The system shall allow users to select a route shared by a followed user.

\item \label{fr-watch:fr31}
The system shall display basic information about routes shared by friends.

\end{enumerate}


\subsection{Interaction \& Smartwatch Constraints}

\begin{enumerate}[label=\textbf{FR\arabic*}, ref=FR\arabic*, resume]

\item \label{fr-watch:fr32}
The system shall present information in a format optimized for a small screen.

\item \label{fr-watch:fr33}
The system shall allow users to navigate content through touch-based scrolling.

\item \label{fr-watch:fr34}
The system shall minimize user interaction steps to access route information.

\item \label{fr-watch:fr35}
The system shall maintain the selected route context during navigation between views.

\end{enumerate}

\subsection{Offline Mode}
\begin{enumerate}[label=\textbf{FR\arabic*}, ref=FR\arabic*, resume]

\item \label{fr-watch:offline-detect}
The system shall detect when the internet connection is not available


\item \label{fr-watch:offline-page}
When the internet connection is not available, the system shall display a dedicated offline page

\end{enumerate}
